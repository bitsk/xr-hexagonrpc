cmake_minimum_required(VERSION 3.16)
project(xr_hexagonrpc VERSION 0.5.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
file(GLOB_RECURSE CLIENT_SOURCES CONFIGURE_DEPENDS libhexagonrpc/*.c)
add_library(xr_hexagonrpc STATIC ${CLIENT_SOURCES})
set_target_properties(xr_hexagonrpc PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME client)
target_include_directories(xr_hexagonrpc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/xr-hexagonrpc>)
install(TARGETS xr_hexagonrpc EXPORT XRHexagonRPCTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/xr-hexagonrpc)
install(EXPORT XRHexagonRPCTargets NAMESPACE XRHexagonRPC:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/XRHexagonRPC)
configure_package_config_file(cmake/XRHexagonRPCConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/XRHexagonRPCConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/XRHexagonRPC)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/XRHexagonRPCConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/XRHexagonRPC)
install(DIRECTORY include libhexagonrpc hexagonrpcd DESTINATION ${CMAKE_INSTALL_DATADIR}/xr-hexagonrpc/source)
install(FILES COPYING DESTINATION ${CMAKE_INSTALL_DATADIR}/xr-hexagonrpc/source)
